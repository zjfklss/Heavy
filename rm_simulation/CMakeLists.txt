cmake_minimum_required(VERSION 3.16)
include(cmake/ProjectLayout.cmake)
project(rm_simulation LANGUAGES CXX)

include(CTest)
option(RM_SIM_BUILD_CLI "Build JSON scenario runners and PNG recording" ON)
option(RM_SIM_BUILD_QT "Build native Qt viewer" ON)

# The simulation packages remain independent of device SDKs and the vision project.
add_subdirectory(modules/world)
add_subdirectory(modules/protocol)
add_subdirectory(modules/camera)
add_subdirectory(modules/actuator)
add_subdirectory(modules/runner)
add_subdirectory(modules/adapters)
if(RM_SIM_BUILD_CLI)
  add_subdirectory(modules/scenario)
  add_subdirectory(modules/recording)
  add_subdirectory(apps/cli)
endif()
if(RM_SIM_BUILD_QT)
  add_subdirectory(apps/viewer)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
