cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(rm_sim_adapter_sdk LANGUAGES CXX)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The TCP adapter SDK currently supports Linux")
endif()
get_filename_component(adapter_root "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT TARGET rm_sim_adapter_profiles)
  add_library(rm_sim_adapter_profiles
    ${adapter_root}/src/profile.cpp
    ${adapter_root}/algorithms/generic/profile.cpp
    ${adapter_root}/algorithms/sp_vision/profile.cpp)
  target_include_directories(rm_sim_adapter_profiles PUBLIC ${adapter_root}/include)
  target_compile_features(rm_sim_adapter_profiles PUBLIC cxx_std_17)
endif()
find_package(nlohmann_json REQUIRED)
add_library(rm_sim_adapter_wire ${adapter_root}/src/wire.cpp server.cpp)
target_include_directories(rm_sim_adapter_wire PUBLIC ${adapter_root}/include)
target_link_libraries(rm_sim_adapter_wire PUBLIC nlohmann_json::nlohmann_json rm_sim_adapter_profiles)
target_compile_features(rm_sim_adapter_wire PUBLIC cxx_std_17)
if(COMMAND rm_sim_enable_warnings)
  rm_sim_enable_warnings(rm_sim_adapter_wire)
endif()
