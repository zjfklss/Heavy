# 仅编重装吊射入口：无 OpenVINO / 相机 / YOLO
# 用法：cmake -S . -B build_lob -C CMakeLists.heavy_lob.txt
# 更好：见根 CMakeLists 末尾 HEAVY_LOB_ONLY / 默认附加目标

cmake_minimum_required(VERSION 3.16.3)
project(heavy_lob_only CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)

find_package(yaml-cpp REQUIRED)
find_package(fmt REQUIRED)

add_subdirectory(tasks/lob_shot)

add_executable(heavy_lob src/heavy_lob.cpp)
target_include_directories(heavy_lob PRIVATE ${CMAKE_SOURCE_DIR}/tasks)
target_link_libraries(heavy_lob PRIVATE lob_shot yaml-cpp fmt::fmt)

add_executable(heavy_lob_offline_test tests/heavy_lob_offline_test.cpp)
target_include_directories(heavy_lob_offline_test PRIVATE ${CMAKE_SOURCE_DIR}/tasks)
target_link_libraries(heavy_lob_offline_test PRIVATE lob_shot)
