cmake_minimum_required(VERSION 3.16.3)
project(sp_vision LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(CTest)
option(BUILD_HERO "Build the hero application" ON)
option(BUILD_CALIBRATION "Build offline calibration tools" ON)
if(NOT BUILD_HERO AND NOT BUILD_CALIBRATION)
  message(FATAL_ERROR "Enable BUILD_HERO or BUILD_CALIBRATION")
endif()

find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)

add_subdirectory(tools)
if(BUILD_CALIBRATION)
  add_subdirectory(calibration)
endif()

if(BUILD_HERO)
  find_package(OpenVINO REQUIRED COMPONENTS Runtime)
  add_subdirectory(io)
  add_subdirectory(tasks/auto_aim)
  add_subdirectory(tasks/auto_outpost)

  add_executable(hero src/hero.cpp)
  target_link_libraries(hero PRIVATE auto_outpost io hero_io)

  if(BUILD_TESTING)
    add_subdirectory(tests/hero)
  endif()
endif()
