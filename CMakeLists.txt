cmake_minimum_required(VERSION 3.16.3)
project(sp_vision LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

include(CTest)
option(BUILD_HERO "Build the hero application" ON)
option(BUILD_CALIBRATION "Build offline calibration tools" ON)
option(HERO_WITH_SIMULATION "Build the rm_simulation TCP sensor/control entry (hero --simulation)" ON)
option(HERO_WITH_SHOW_UI "Build the Qt5 diagnostic dashboard (hero --show)" ON)
set(HERO_SIM_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/io/simulation/protocol/sdk/cpp" CACHE PATH "RM simulator protocol SDK")
if(NOT BUILD_HERO AND NOT BUILD_CALIBRATION)
  message(FATAL_ERROR "Enable BUILD_HERO or BUILD_CALIBRATION")
endif()

find_package(OpenCV REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(fmt REQUIRED)
find_package(spdlog REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(Threads REQUIRED)

add_subdirectory(tools)
if(BUILD_CALIBRATION)
  add_subdirectory(calibration)
endif()

if(BUILD_HERO)
  find_package(OpenVINO REQUIRED COMPONENTS Runtime)
  add_subdirectory(io)
  add_subdirectory(tasks/auto_aim)
  add_subdirectory(tasks/auto_outpost)

  # tools/jlcv_show is kept identical to sp_jlcv_aimer; it reads JLCV_WITH_SHOW_UI.
  set(JLCV_WITH_SHOW_UI ${HERO_WITH_SHOW_UI})
  if(HERO_WITH_SHOW_UI)
    find_package(Qt5 COMPONENTS Widgets REQUIRED)
  endif()
  add_subdirectory(tools/jlcv_show)

  add_executable(hero src/hero.cpp)
  target_link_libraries(hero PRIVATE auto_outpost io hero_io jlcv_show_model)
  if(HERO_WITH_SHOW_UI)
    target_compile_definitions(hero PRIVATE HERO_WITH_SHOW_UI)
    target_link_libraries(hero PRIVATE jlcv_show)
  endif()

  if(HERO_WITH_SIMULATION)
    if(NOT EXISTS "${HERO_SIM_SDK_DIR}/CMakeLists.txt")
      message(FATAL_ERROR "Set HERO_SIM_SDK_DIR to the simulator's modules/adapters/sdk/cpp")
    endif()
    add_subdirectory("${HERO_SIM_SDK_DIR}" "${CMAKE_BINARY_DIR}/simulation_protocol")
    add_subdirectory(io/simulation)
    target_compile_definitions(hero PRIVATE HERO_WITH_SIMULATION)
  endif()
  add_subdirectory(src/simulation)
  target_link_libraries(hero PRIVATE hero_simulation_launcher)
  if(HERO_WITH_SIMULATION)
    target_link_libraries(hero PRIVATE hero_simulation)
  endif()

  if(BUILD_TESTING)
    add_subdirectory(tests/hero)
  endif()
endif()
